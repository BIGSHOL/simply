## 1. 심리학적 프레임워크

### 1.1 컨셉

조선시대에 환생한다면 어떤 신분/직업이 되었을까? 사극 드라마 감성을 MZ식으로 풀어낸 재미 테스트. 현대의 일상 상황을 조선시대 세계관으로 치환하여 몰입감과 유머를 동시에 잡는다.

### 1.2 측정 차원 정의 (3축)

| 차원 | 코드 | 정의 | 낮음 (0-3) | 높음 (4-6) |
|------|------|------|------------|------------|
| **주도성 (Leadership)** | `leadership` | 앞에 나서서 이끌고 결정하는 성향 vs 뒤에서 보좌하고 따르는 성향 | 팔로워십: 보좌형, 뒤에서 묵묵히 | 리더십: 주도형, 앞장서서 이끄는 |
| **안정추구 (Stability)** | `stability` | 안정적이고 예측 가능한 것을 선호 vs 새로운 도전과 모험 선호 | 모험형: 변화, 자유, 새로움 추구 | 안정형: 루틴, 안전, 체계 추구 |
| **사회성 (Sociability)** | `sociability` | 혼자만의 시간과 공간 선호 vs 사람들과 어울리는 것 선호 | 독립형: 혼자, 개인 활동, 내향 | 사교형: 어울림, 네트워킹, 외향 |

### 1.3 차원 조합과 유형 분류 원리

3개 차원의 High(H)/Low(L) 조합으로 8가지 기본 유형이 가능하며, 점수 세분화를 통해 10개 유형으로 분류합니다.

```
유형 분류 로직:
1. 3개 차원 점수를 각각 0~6점으로 집계 (문항당 +2 또는 +0)
2. 주도성/안정추구/사회성의 H/L 조합으로 1차 분류
3. 극단값(0 또는 6)과 중간값(2-4)을 활용한 세분화로 10개 유형 확정
```

### 4.4 유형 분류 알고리즘

```typescript
type JoseonType =
  | 'king'              // 01. 임금
  | 'crown_prince'      // 02. 세자
  | 'hanryang'          // 03. 한량
  | 'royal_doctor'      // 04. 어의
  | 'merchant'          // 05. 보부상
  | 'gisaeng'           // 06. 기생
  | 'military_officer'  // 07. 무관
  | 'eccentric'         // 08. 기인
  | 'nobleman'          // 09. 도령/아씨
  | 'court_lady';       // 10. 궁녀

interface DimensionScores {
  leadership: number;    // 0-6
  stability: number;     // 0-8
  sociability: number;   // 0-6
}

/**
 * 조선시대 환생 신분 결정 알고리즘
 *
 * 3축 점수 조합으로 10개 유형 분류
 * - 주도성: HIGH(4+) / LOW(0-3)
 * - 안정추구: HIGH(5+) / MID(3-4) / LOW(0-2)
 * - 사회성: HIGH(4+) / LOW(0-3)
 */
function determineJoseonType(scores: DimensionScores): JoseonType {
  const { leadership, stability, sociability } = scores;

  // 주도성 분류
  const isLeader = leadership >= 4;       // HIGH: 리더형
  const isFollower = leadership <= 2;      // LOW: 팔로워형
  // leadership 3 = 중간

  // 안정추구 분류
  const isStable = stability >= 5;         // HIGH: 안정형
  const isAdventurer = stability <= 2;     // LOW: 모험형
  // stability 3-4 = 중간

  // 사회성 분류
  const isSocial = sociability >= 4;       // HIGH: 사교형
  const isIndependent = sociability <= 2;  // LOW: 독립형
  // sociability 3 = 중간

  // =============================================
  // 1. 주도성 HIGH (리더형) 유형들
  // =============================================
  if (isLeader) {
    // 리더 + 안정 + 사교 = 임금 (모든 것을 갖춘 왕)
    if (isStable && isSocial) return 'king';

    // 리더 + 모험 + 사교 = 보부상 (이끌면서 돌아다니며 사람 만남)
    if (isAdventurer && isSocial) return 'merchant';

    // 리더 + 모험 + 독립 = 무관 (홀로 전장을 누비는 용장)
    if (isAdventurer && isIndependent) return 'military_officer';

    // 리더 + 안정 + 독립 = 어의 (묵묵히 전문성으로 이끔)
    if (isStable && isIndependent) return 'royal_doctor';

    // 리더 + 중간 조합 = 세자 (아직 완성되진 않았지만 리더 자질)
    return 'crown_prince';
  }

  // =============================================
  // 2. 주도성 LOW (팔로워형) 유형들
  // =============================================
  if (isFollower) {
    // 팔로워 + 안정 + 독립 = 궁녀 (뒤에서 묵묵히 체계적으로)
    if (isStable && isIndependent) return 'court_lady';

    // 팔로워 + 안정 + 사교 = 도령/아씨 (반듯하고 사교적인 양반)
    if (isStable && isSocial) return 'nobleman';

    // 팔로워 + 모험 + 독립 = 기인 (혼자 남다른 길을 가는 4차원)
    if (isAdventurer && isIndependent) return 'eccentric';

    // 팔로워 + 모험 + 사교 = 기생 (자유로운 영혼의 엔터테이너)
    if (isAdventurer && isSocial) return 'gisaeng';

    // 팔로워 + 중간 조합 = 도령/아씨 (기본적으로 반듯한 성향)
    return 'nobleman';
  }

  // =============================================
  // 3. 주도성 MID (중간) 유형들
  // =============================================
  // 중간 리더십 + 모험 + 사교 = 기생 (엔터테이너형)
  if (isAdventurer && isSocial) return 'gisaeng';

  // 중간 리더십 + 모험 + 독립 = 한량 (자유로운 풍류객)
  if (isAdventurer && isIndependent) return 'hanryang';

  // 중간 리더십 + 안정 + 사교 = 도령/아씨
  if (isStable && isSocial) return 'nobleman';

  // 중간 리더십 + 안정 + 독립 = 궁녀
  if (isStable && isIndependent) return 'court_lady';

  // 중간 리더십 + 모험 + 중간 사회성 = 한량 (풍류의 길)
  if (isAdventurer) return 'hanryang';

  // 중간 리더십 + 안정 + 중간 사회성 = 어의 (전문가의 길)
  if (isStable) return 'royal_doctor';

  // 모든 차원 중간 = 세자 (아직 가능성이 열려있는)
  return 'crown_prince';
}
```

```typescript
 * 답변 배열로부터 차원별 점수 집계

function calculateScores(answers: Answer[]): DimensionScores {
  const scores: DimensionScores = {
    leadership: 0,
    stability: 0,
    sociability: 0,
  };

  answers.forEach((answer) => {
    const question = scoringTable.find((q) => q.questionId === answer.questionId);
    if (!question) return;

    const choiceScore = answer.choiceId === 'A' ? question.choiceA : question.choiceB;
    scores.leadership += choiceScore.leadership;
    scores.stability += choiceScore.stability;
    scores.sociability += choiceScore.sociability;
  });

  return scores;
}

function getJoseonResult(answers: Answer[]): JoseonType {
  const scores = calculateScores(answers);
  return determineJoseonType(scores);
```

## 6. 문항 설계 근거

### 6.1 측정 영역 분포

| 측정 차원 | 문항 번호 | 개수 |
|----------|----------|------|
| 주도성 (leadership) | Q1, Q4, Q7 | 3개 |
| 안정추구 (stability) | Q2, Q5, Q8, Q10 | 4개 |
| 사회성 (sociability) | Q3, Q6, Q9 | 3개 |

> 안정추구 차원이 4문항으로 가장 많은 이유: 이 차원이 유형 분류에서 가장 넓은 범위(0-8)를 갖고, HIGH/MID/LOW 3단계로 세분화되어 더 세밀한 분류가 필요하기 때문.

### 6.2 문항 설계 원칙

1. **현대-조선 치환**: 모든 상황은 현대 일상을 조선시대 용어로 치환 (괄호 안에 현대 의미 병기)
2. **사극 밈 감성**: 사극 드라마에서 볼 법한 표현을 MZ식 유머로 재해석
3. **양쪽 모두 매력적**: A/B 어느 쪽을 골라도 부정적이지 않은 선택지 설계
4. **직관적 이해**: 응답자가 자신의 성향을 바로 떠올릴 수 있는 구체적 상황 제시

### 6.3 심리측정학적 고려사항

1. **내적 일관성**: 각 차원을 3-4개 문항으로 반복 측정하여 신뢰도 확보
2. **구성 타당도**: 주도성/안정추구/사회성이라는 성격심리학의 핵심 차원 활용
3. **안면 타당도**: 조선시대 세계관이라는 재미 요소 속에서도 성향을 직관적으로 측정
4. **사회적 바람직성 최소화**: 어느 쪽이든 긍정적인 이미지의 선택지로 구성

### 7.1 참여 동기 유발
- **사극 세계관 몰입**: "그대는 조선에 환생하였도다!" 도입부로 분위기 세팅
- **MZ 공감**: 현대 상황의 조선 치환이 공감과 웃음 유발
- **자기 발견**: "내가 조선에 태어났으면 뭐 했을까?" 호기심 자극

### 7.2 공유 촉진 요소
- **캐치한 별명**: "조선의 워라밸 장인", "조선의 CEO" 등 SNS 공유 욕구 자극
- **해시태그**: 바로 복붙 가능한 재미있는 해시태그 제공
- **현대 직업 매칭**: "조선시대엔 한량, 현대엔 프리랜서" 같은 비교 포인트
- **궁합 기능**: "너는 뭐 나왔어?" 친구에게 공유하고 비교하는 재미
- **사극 밈**: "어명이다", "과인이 틀렸느냐" 등 밈화 가능한 요소 포함

### 7.3 톤앤매너 가이드

- 과하지 않은 유머 (사극 패러디 수준)
